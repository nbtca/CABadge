cmake_minimum_required(VERSION 3.20)
project(JXBadgeSimulator C)
set(IDF_JSON "$ENV{USERPROFILE}/.platformio/packages/framework-espidf/components/json/cJSON")
set(CMAKE_C_STANDARD 11)
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS vendor/lvgl-9.4.0/src/*.c)
add_library(lvgl OBJECT ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC vendor/lvgl-9.4.0 ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_executable(JXBadgeSimulator simulator.c bridge.c "${IDF_JSON}/cJSON.c" ui/badge_ui.c ui/wifi_panel.c ui/assets.c)
target_include_directories(JXBadgeSimulator PRIVATE "${IDF_JSON}" vendor/SDL2-2.32.10/x86_64-w64-mingw32/include/SDL2)
target_compile_definitions(JXBadgeSimulator PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(JXBadgeSimulator PRIVATE lvgl "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL2-2.32.10/x86_64-w64-mingw32/lib/libSDL2.dll.a")
add_custom_command(TARGET JXBadgeSimulator POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL2-2.32.10/x86_64-w64-mingw32/bin/SDL2.dll" "$<TARGET_FILE_DIR:JXBadgeSimulator>")
enable_testing()
add_test(NAME round_layout COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/.venv/Scripts/python.exe" "${CMAKE_CURRENT_SOURCE_DIR}/check_layout.py")
add_test(NAME ui_interaction COMMAND JXBadgeSimulator --self-test)
set_tests_properties(ui_interaction PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(usb_telemetry_test usb_telemetry.c)
target_compile_definitions(usb_telemetry_test PRIVATE USB_TELEMETRY_TEST)
add_test(NAME usb_telemetry COMMAND usb_telemetry_test)
add_executable(usb_screen_protocol_test usb_screen/protocol_test.c)
add_test(NAME usb_screen_protocol COMMAND usb_screen_protocol_test)
add_executable(shake_test usb_screen/shake_test.c)
add_test(NAME shake_detector COMMAND shake_test)
add_executable(wallpaper_store_test usb_screen/wallpaper_store.c usb_screen/wallpaper_store_test.c)
add_test(NAME wallpaper_storage COMMAND wallpaper_store_test)
add_executable(management_test usb_screen/management_test.c)
add_test(NAME management_protocol COMMAND management_test)
# Compile the production service logic against host I/O substitutes, not a copied implementation.
set(MOCK_ESP "${CMAKE_CURRENT_BINARY_DIR}/service_test_include")
foreach(header esp_flash.h esp_err.h esp_timer.h esp_random.h esp_partition.h esp_heap_caps.h esp_http_server.h wallpaper_html.h freertos/FreeRTOS.h freertos/semphr.h freertos/queue.h freertos/task.h ui/badge_ui.h src/misc/cache/lv_cache.h)
    get_filename_component(dir "${MOCK_ESP}/${header}" DIRECTORY)
    file(MAKE_DIRECTORY "${dir}")
    file(WRITE "${MOCK_ESP}/${header}" "/* Types supplied by service_test_shim.h. */\n")
endforeach()
set(IDF_JSON "$ENV{USERPROFILE}/.platformio/packages/framework-espidf/components/json/cJSON")
add_executable(management_service_test usb_screen/service_test.c usb_screen/wallpaper_store.c "${IDF_JSON}/cJSON.c")
target_include_directories(management_service_test PRIVATE "${MOCK_ESP}" usb_screen "${IDF_JSON}")
add_test(NAME management_services COMMAND management_service_test)

add_executable(perf_stats_test usb_screen/perf_stats_test.c)
add_test(NAME perf_statistics COMMAND perf_stats_test)

add_executable(render_benchmark render_benchmark.c ui/wifi_panel.c ui/assets.c)
target_link_libraries(render_benchmark PRIVATE lvgl)
add_test(NAME render_cache COMMAND render_benchmark)
